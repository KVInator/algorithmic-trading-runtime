function(add_trading_contract target test_name libraries)
    add_executable(${target} ${target}.cpp)
    target_link_libraries(${target} PRIVATE ${libraries})
    target_compile_definitions(${target} PRIVATE TRADING_PROJECT_ROOT="${PROJECT_SOURCE_DIR}")
    configure_trading_target(${target})
    if(ARGV3)
        add_test(NAME ${test_name} COMMAND ${target} "${PROJECT_BINARY_DIR}/${target}-output")
    else()
        add_test(NAME ${test_name} COMMAND ${target})
    endif()
endfunction()
